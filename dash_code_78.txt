import { useState } from 'react';
import { useNavigate } from 'react-router-dom';

const initialDeals = [
  { id: 'DEAL-104', company: 'Acme Global Corp', title: 'Global Multi-Region Cloud Migration', value: 380000, stage: 'Negotiation', contact: 'sarah.c@acmeglobal.com', owner: 'Alex M.', probability: 85 },
  { id: 'DEAL-105', company: 'NovaTech FinTech', title: 'AI Risk Assessment Engine License', value: 245000, stage: 'Proposal', contact: 'marcus.v@novatech.io', owner: 'Jordan K.', probability: 60 },
  { id: 'DEAL-106', company: 'Starlight Retail', title: 'Omnichannel POS & Analytics Hub', value: 520000, stage: 'Closed Won', contact: 'elena.r@starlight.com', owner: 'Alex M.', probability: 100 },
  { id: 'DEAL-107', company: 'HyperScale AI', title: 'Enterprise Dedicated GPU Orchestration', value: 950000, stage: 'Discovery', contact: 'chen.w@hyperscale.ai', owner: 'David L.', probability: 35 },
  { id: 'DEAL-108', company: 'AeroDynamics Lab', title: 'Autonomous Simulation Suite', value: 410000, stage: 'Negotiation', contact: 'viktor.p@aerolab.org', owner: 'Elena S.', probability: 75 },
];

  const [searchQuery, setSearchQuery] = useState('');
  const [stageFilter, setStageFilter] = useState('ALL');
  const [activeTab, setActiveTab] = useState('table'); // 'table' or 'kanban'
  const [isModalOpen, setIsModalOpen] = useState(false);

  const [newCompany, setNewCompany] = useState('');
  const [newTitle, setNewTitle] = useState('');
  const [newValue, setNewValue] = useState('');
  const [newStage, setNewStage] = useState('Discovery');
  const [newContact, setNewContact] = useState('');

  const handleLogout = () => {
    setIsAuthenticated(false);
    navigate('/login');
  };

    e.preventDefault();
    if (!newCompany || !newTitle || !newValue) return;

    const newDeal = {
      id: `DEAL-${Math.floor(100 + Math.random() * 900)}`,
      company: newCompany,
      title: newTitle,
      value: parseFloat(newValue) || 0,
      stage: newStage,
      contact: newContact || 'contact@client.com',
      owner: 'Alex M.',
      probability: newStage === 'Closed Won' ? 100 : newStage === 'Negotiation' ? 80 : newStage === 'Proposal' ? 60 : 30,
    };

    setDeals([newDeal, ...deals]);
    setNewCompany('');
    setNewTitle('');
    setNewValue('');
    setNewContact('');
    setIsModalOpen(false);
  };

  const deleteDeal = (id) => {
    setDeals(deals.filter((d) => d.id !== id));
  };

  // Filtered deals
  const filteredDeals = deals.filter((deal) => {
    const matchesSearch =
      deal.company.toLowerCase().includes(searchQuery.toLowerCase()) ||
      deal.title.toLowerCase().includes(searchQuery.toLowerCase()) ||
      deal.id.toLowerCase().includes(searchQuery.toLowerCase());
    const matchesStage = stageFilter === 'ALL' || deal.stage === stageFilter;
    return matchesSearch && matchesStage;
  });

  // Calculate totals
  const totalPipeline = deals.reduce((acc, d) => acc + d.value, 0);
  const closedWonTotal = deals.filter((d) => d.stage === 'Closed Won').reduce((acc, d) => acc + d.value, 0);

  const getStageBadge = (stage) => {
    switch (stage) {
      case 'Closed Won':
        return 'bg-emerald-500/15 text-emerald-300 border-emerald-500/30';
      case 'Negotiation':
        return 'bg-amber-500/15 text-amber-300 border-amber-500/30';
      case 'Proposal':
        return 'bg-cyan-500/15 text-cyan-300 border-cyan-500/30';
      default:
        return 'bg-indigo-500/15 text-indigo-300 border-indigo-500/30';
    }
  };

  return (
    <div className="min-h-screen bg-[#080c14] text-slate-100 font-sans flex flex-col selection:bg-cyan-500 selection:text-white">
      {/* Top Navigation */}
      <header className="sticky top-0 z-30 bg-[#080c14]/80 backdrop-blur-xl border-b border-slate-800/80 px-4 sm:px-8 py-3.5">
        <div className="max-w-7xl mx-auto flex items-center justify-between gap-4">
          {/* Logo & Workspace */}
          <div className="flex items-center gap-4">
            <div className="flex items-center gap-2.5">
              <div className="w-9 h-9 rounded-xl bg-gradient-to-tr from-cyan-500 via-blue-600 to-indigo-600 flex items-center justify-center shadow-md shadow-cyan-500/20 ring-1 ring-white/20">
                <svg className="w-5 h-5 text-white" fill="none" stroke="currentColor" strokeWidth="2.5" viewBox="0 0 24 24">
                  <path strokeLinecap="round" strokeLinejoin="round" d="M13 10V3L4 14h7v7l9-11h-7z" />
                </svg>
              </div>
              <span className="text-xl font-bold font-display tracking-tight text-white hidden sm:inline-block">
                Deals<span className="text-transparent bg-clip-text bg-gradient-to-r from-cyan-400 to-blue-400">Flow</span>
              </span>
            </div>

            <div className="h-5 w-px bg-slate-800 hidden sm:block" />

            <div className="flex items-center gap-2 px-3 py-1 rounded-lg bg-slate-900/80 border border-slate-800 text-xs text-slate-300">
              <span className="w-2 h-2 rounded-full bg-emerald-400 animate-pulse" />
              <span className="font-medium">Acme Ventures Production</span>
            </div>
          </div>

          {/* Right actions */}
          <div className="flex items-center gap-3">
            <button
              onClick={() => setIsModalOpen(true)}
              className="py-2 px-3.5 rounded-xl font-semibold text-xs text-white bg-gradient-to-r from-cyan-500 to-blue-600 hover:from-cyan-400 hover:to-blue-500 shadow-md shadow-cyan-500/20 flex items-center gap-1.5 transition-all transform active:scale-95 cursor-pointer"
            >
              <svg className="w-4 h-4" fill="none" stroke="currentColor" strokeWidth="2.5" viewBox="0 0 24 24">
                <path strokeLinecap="round" strokeLinejoin="round" d="M12 4.5v15m7.5-7.5h-15" />
              </svg>
              <span className="hidden sm:inline">Add Deal</span>
            </button>

            {/* User Pill & Logout */}
            <div className="flex items-center gap-2.5 pl-2 border-l border-slate-800">
              <div className="w-8 h-8 rounded-full bg-gradient-to-tr from-indigo-500 to-cyan-400 p-0.5 shadow-sm">
                <div className="w-full h-full rounded-full bg-slate-900 flex items-center justify-center text-xs font-bold text-cyan-300">
                  AM
                </div>
              </div>
              <div className="hidden md:block text-left">
                <p className="text-xs font-semibold text-white leading-tight">Alex Morgan</p>
                <p className="text-[10px] text-slate-400">Enterprise Admin</p>
              </div>
              <button
                onClick={handleLogout}
                className="p-2 rounded-lg text-slate-400 hover:text-rose-400 hover:bg-rose-500/10 border border-transparent hover:border-rose-500/20 transition-all ml-1 cursor-pointer"
                title="Log out"
              >
                <svg className="w-4 h-4" fill="none" stroke="currentColor" strokeWidth="2" viewBox="0 0 24 24">
                  <path strokeLinecap="round" strokeLinejoin="round" d="M15.75 9V5.25A2.25 2.25 0 0013.5 3h-6a2.25 2.25 0 00-2.25 2.25v13.5A2.25 2.25 0 007.5 21h6a2.25 2.25 0 002.25-2.25V15M12 9l-3 3m0 0l3 3m-3-3h12.75" />
                </svg>
              </button>
            </div>
          </div>
        </div>
      </header>

      {/* Main Content Area */}
      <main className="flex-1 max-w-7xl w-full mx-auto px-4 sm:px-8 py-8 space-y-8">
        {/* Executive Banner */}
        <div className="relative rounded-3xl overflow-hidden glass-card p-6 sm:p-8 border border-slate-700/60 bg-gradient-to-r from-slate-900/90 via-slate-900/60 to-blue-950/40">
          <div className="relative z-10 flex flex-col md:flex-row md:items-center justify-between gap-6">
            <div>
              <div className="inline-flex items-center gap-2 px-3 py-1 rounded-full bg-cyan-500/10 border border-cyan-500/20 text-cyan-300 text-xs font-semibold mb-3">
                <span>⚡ Live Deal Velocity</span>
                <span className="w-1.5 h-1.5 rounded-full bg-cyan-400 animate-ping" />
              </div>
              <h1 className="text-2xl sm:text-3xl font-extrabold text-white font-display">
                Executive Revenue Pipeline
              </h1>
              <p className="mt-1 text-sm text-slate-400 max-w-xl">
                Real-time contract valuation, stage forecasts, and active client deal progression.
              </p>
            </div>

            <div className="flex items-center gap-3">
              <button
                onClick={() => setIsModalOpen(true)}
                className="py-2.5 px-4 rounded-xl font-medium text-xs text-white bg-blue-600 hover:bg-blue-500 shadow-lg shadow-blue-500/20 transition-all flex items-center gap-2 cursor-pointer"
              >
                <span>+ Create Pipeline Deal</span>
              </button>
            </div>
          </div>
        </div>

        {/* 4 KPI Metric Cards */}
        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4 sm:gap-5">
          {/* Card 1: Pipeline */}
          <div className="glass-card glass-card-hover rounded-2xl p-5 border border-slate-800">
            <div className="flex items-center justify-between text-slate-400 mb-3">
              <span className="text-xs font-medium uppercase tracking-wider">Active Pipeline</span>
              <span className="p-2 rounded-xl bg-cyan-500/10 text-cyan-400">
                <svg className="w-4 h-4" fill="none" stroke="currentColor" strokeWidth="2" viewBox="0 0 24 24">
                  <path strokeLinecap="round" strokeLinejoin="round" d="M12 6v12m-3-2.818l.879.659c1.171.879 3.07.879 4.242 0 1.172-.879 1.172-2.303 0-3.182C13.536 12.219 12.768 12 12 12c-.725 0-1.45-.22-2.003-.659-1.106-.879-1.106-2.303 0-3.182s2.9-.879 4.006 0l.415.33M21 12a9 9 0 11-18 0 9 9 0 0118 0z" />
                </svg>
              </span>
            </div>
            <div className="text-2xl sm:text-3xl font-extrabold text-white font-display">
              ${(totalPipeline / 1000000).toFixed(2)}M
            </div>
            <div className="mt-2.5 flex items-center gap-1.5 text-xs text-emerald-400">
              <svg className="w-3.5 h-3.5" fill="none" stroke="currentColor" strokeWidth="2.5" viewBox="0 0 24 24">
                <path strokeLinecap="round" strokeLinejoin="round" d="M4.5 10.5L12 3m0 0l7.5 7.5M12 3v18" />
              </svg>
              <span className="font-semibold">+18.4%</span>
              <span className="text-slate-400">vs last month</span>
            </div>
          </div>

          {/* Card 2: Closed Deals */}
          <div className="glass-card glass-card-hover rounded-2xl p-5 border border-slate-800">
            <div className="flex items-center justify-between text-slate-400 mb-3">
              <span className="text-xs font-medium uppercase tracking-wider">Closed Won</span>
              <span className="p-2 rounded-xl bg-emerald-500/10 text-emerald-400">
                <svg className="w-4 h-4" fill="none" stroke="currentColor" strokeWidth="2" viewBox="0 0 24 24">
                  <path strokeLinecap="round" strokeLinejoin="round" d="M9 12.75L11.25 15 15 9.75M21 12a9 9 0 11-18 0 9 9 0 0118 0z" />
                </svg>
              </span>
            </div>
            <div className="text-2xl sm:text-3xl font-extrabold text-white font-display">
              ${(closedWonTotal / 1000).toFixed(0)}k
            </div>
            <div className="mt-2.5 flex items-center gap-1.5 text-xs text-emerald-400">
              <svg className="w-3.5 h-3.5" fill="none" stroke="currentColor" strokeWidth="2.5" viewBox="0 0 24 24">
                <path strokeLinecap="round" strokeLinejoin="round" d="M4.5 10.5L12 3m0 0l7.5 7.5M12 3v18" />
              </svg>
              <span className="font-semibold">+24.0%</span>
              <span className="text-slate-400">closed revenue</span>
            </div>
          </div>

          {/* Card 3: Win Rate */}
          <div className="glass-card glass-card-hover rounded-2xl p-5 border border-slate-800">
            <div className="flex items-center justify-between text-slate-400 mb-3">
              <span className="text-xs font-medium uppercase tracking-wider">Win Rate</span>
              <span className="p-2 rounded-xl bg-indigo-500/10 text-indigo-400">
                <svg className="w-4 h-4" fill="none" stroke="currentColor" strokeWidth="2" viewBox="0 0 24 24">
                  <path strokeLinecap="round" strokeLinejoin="round" d="M3.75 13.5l10.5-11.25L12 10.5h8.25L9.75 21.75 12 13.5H3.75z" />
                </svg>
              </span>
            </div>
            <div className="text-2xl sm:text-3xl font-extrabold text-white font-display">
              68.4%
            </div>
            <div className="mt-2.5 flex items-center gap-1.5 text-xs text-cyan-400">
              <svg className="w-3.5 h-3.5" fill="none" stroke="currentColor" strokeWidth="2.5" viewBox="0 0 24 24">
                <path strokeLinecap="round" strokeLinejoin="round" d="M4.5 10.5L12 3m0 0l7.5 7.5M12 3v18" />
              </svg>
              <span className="font-semibold">+4.2%</span>
              <span className="text-slate-400">benchmark index</span>
            </div>
          </div>

          {/* Card 4: Deal Velocity */}
          <div className="glass-card glass-card-hover rounded-2xl p-5 border border-slate-800">
            <div className="flex items-center justify-between text-slate-400 mb-3">
              <span className="text-xs font-medium uppercase tracking-wider">Avg Sales Cycle</span>
              <span className="p-2 rounded-xl bg-amber-500/10 text-amber-400">
                <svg className="w-4 h-4" fill="none" stroke="currentColor" strokeWidth="2" viewBox="0 0 24 24">
                  <path strokeLinecap="round" strokeLinejoin="round" d="M12 6v6h4.5m4.5 0a9 9 0 11-18 0 9 9 0 0118 0z" />
                </svg>
              </span>
            </div>
            <div className="text-2xl sm:text-3xl font-extrabold text-white font-display">
              18 Days
            </div>
            <div className="mt-2.5 flex items-center gap-1.5 text-xs text-emerald-400">
              <span className="font-semibold">-3.5 days</span>
              <span className="text-slate-400">accelerated velocity</span>
            </div>
          </div>
        </div>

        {/* Deals Table & Controls */}
        <div className="glass-card rounded-3xl border border-slate-800/80 overflow-hidden shadow-2xl">
          {/* Controls Bar */}
          <div className="p-5 sm:p-6 border-b border-slate-800 flex flex-col md:flex-row items-center justify-between gap-4">
            {/* Search */}
            <div className="relative w-full md:w-80">
              <div className="absolute inset-y-0 left-0 pl-3.5 flex items-center pointer-events-none text-slate-400">
                <svg className="w-4 h-4" fill="none" stroke="currentColor" strokeWidth="2" viewBox="0 0 24 24">
                  <path strokeLinecap="round" strokeLinejoin="round" d="M21 21l-5.197-5.197m0 0A7.5 7.5 0 105.196 5.196a7.5 7.5 0 0010.607 10.607z" />
                </svg>
              </div>
              <input
                type="text"
                value={searchQuery}
                onChange={(e) => setSearchQuery(e.target.value)}
                placeholder="Search deals, company or ID..."
                className="w-full py-2 pl-10 pr-4 rounded-xl glass-input text-xs placeholder-slate-500 outline-none"
              />
            </div>

            {/* Filter buttons */}
            <div className="flex items-center gap-2 overflow-x-auto w-full md:w-auto pb-1 md:pb-0">
              {['ALL', 'Discovery', 'Proposal', 'Negotiation', 'Closed Won'].map((stage) => (
                <button
                  key={stage}
                  onClick={() => setStageFilter(stage)}
                  className={`px-3 py-1.5 rounded-xl text-xs font-medium transition-all whitespace-nowrap cursor-pointer ${
                    stageFilter === stage
                      ? 'bg-cyan-500/20 text-cyan-300 border border-cyan-500/40'
                      : 'bg-slate-900/60 text-slate-400 hover:text-slate-200 border border-slate-800'
                  }`}
                >
                  {stage}
                </button>
              ))}
            </div>
          </div>

          {/* Table */}
          <div className="overflow-x-auto">
            <table className="w-full text-left text-sm text-slate-300">
              <thead className="bg-slate-900/80 text-[11px] font-semibold text-slate-400 uppercase tracking-wider border-b border-slate-800">
                <tr>
                  <th className="py-3.5 px-6">Deal & Company</th>
                  <th className="py-3.5 px-6">Stage</th>
                  <th className="py-3.5 px-6">Contract Value</th>
                  <th className="py-3.5 px-6">Probability</th>
                  <th className="py-3.5 px-6">Owner</th>
                  <th className="py-3.5 px-6 text-right">Actions</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-slate-800/60 font-sans">
                {filteredDeals.length > 0 ? (
                  filteredDeals.map((deal) => (
                    <tr key={deal.id} className="hover:bg-slate-800/30 transition-colors">
                      <td className="py-4 px-6">
                        <div className="flex items-center gap-3">
                          <div className="w-9 h-9 rounded-xl bg-gradient-to-tr from-slate-800 to-slate-700 border border-slate-700 flex items-center justify-center text-xs font-bold text-slate-200">
                            {deal.company.slice(0, 2).toUpperCase()}
                          </div>
                          <div>
                            <div className="font-semibold text-white text-sm">{deal.title}</div>
                            <div className="text-xs text-slate-400 flex items-center gap-1.5 mt-0.5">
                              <span>{deal.company}</span>
                              <span>•</span>
                              <span className="text-[10px] text-slate-500">{deal.id}</span>
                            </div>
                          </div>
                        </div>
                      </td>
                      <td className="py-4 px-6">
                        <span className={`inline-flex items-center px-2.5 py-1 rounded-full text-xs font-medium border ${getStageBadge(deal.stage)}`}>
                          {deal.stage}
                        </span>
                      </td>
                      <td className="py-4 px-6 font-semibold text-white">
                        ${deal.value.toLocaleString()}
                      </td>
                      <td className="py-4 px-6">
                        <div className="flex items-center gap-2">
                          <div className="w-16 h-1.5 rounded-full bg-slate-800 overflow-hidden">
                            <div
                              className="h-full rounded-full bg-gradient-to-r from-cyan-400 to-blue-500"
                              style={{ width: `${deal.probability}%` }}
                            />
                          </div>
                          <span className="text-xs text-slate-400">{deal.probability}%</span>
                        </div>

                      <td className="py-4 px-6 text-xs text-slate-300">
                        {deal.owner}
                      </td>
                      <td className="py-4 px-6 text-right">

                          onClick={() => deleteDeal(deal.id)}
                          className="p-1.5 rounded-lg text-slate-500 hover:text-rose-400 hover:bg-rose-500/10 transition-colors cursor-pointer"

                        >
                          <svg className="w-4 h-4" fill="none" stroke="currentColor" strokeWidth="2" viewBox="0 0 24 24">

                          </svg>
                        </button>
                      </td>
                    </tr>
                  ))
                ) : (
                  <tr>
                    <td colSpan="6" className="py-12 text-center text-slate-500 text-xs">
                      No deals match your active search or filter criteria.
                    </td>
                  </tr>
                )}
              </tbody>
            </table>
          </div>
        </div>
      </main>

      {/* Modal: Add New Deal */}
      {isModalOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/70 backdrop-blur-md">
          <div className="relative w-full max-w-md glass-card rounded-3xl p-6 sm:p-8 border border-slate-700 shadow-2xl animate-float">

              <h3 className="text-xl font-bold font-display text-white">Create New Deal</h3>

                className="p-1 rounded-lg text-slate-400 hover:text-white hover:bg-slate-800 transition-colors"

                <svg className="w-5 h-5" fill="none" stroke="currentColor" strokeWidth="2" viewBox="0 0 24 24">
                  <path strokeLinecap="round" strokeLinejoin="round" d="M6 18L18 6M6 6l12 12" />
                </svg>
              </button>
            </div>

            <form onSubmit={handleCreateDeal} className="space-y-4">
              <div>
                <label className="block text-xs font-semibold text-slate-300 uppercase tracking-wider mb-1">
                  Company Name
                </label>
                <input
                  type="text"
                  value={newCompany}
                  onChange={(e) => setNewCompany(e.target.value)}
                  placeholder="e.g. Apex Global Solutions"
                  required
                  className="w-full py-2.5 px-3.5 rounded-xl glass-input text-sm outline-none"
                />
              </div>

              <div>
                <label className="block text-xs font-semibold text-slate-300 uppercase tracking-wider mb-1">
                  Deal Title / Project
                </label>
                <input
                  type="text"
                  value={newTitle}
                  onChange={(e) => setNewTitle(e.target.value)}

                  required
                  className="w-full py-2.5 px-3.5 rounded-xl glass-input text-sm outline-none"
                />
              </div>

              <div className="grid grid-cols-2 gap-3">
                <div>
                  <label className="block text-xs font-semibold text-slate-300 uppercase tracking-wider mb-1">
                    Value ($ USD)
                  </label>
                  <input
                    type="number"
                    value={newValue}
                    onChange={(e) => setNewValue(e.target.value)}

                  >

                    <option value="Closed Won">Closed Won</option>

                <button
                  type="button"
                  onClick={() => setIsModalOpen(false)}
                  className="px-4 py-2.5 rounded-xl text-xs font-medium text-slate-400 hover:text-slate-200 transition-colors"
                >
                  Cancel
                </button>
                <button
                  type="submit"
                  className="px-5 py-2.5 rounded-xl text-xs font-semibold text-white bg-gradient-to-r from-cyan-500 to-blue-600 hover:from-cyan-400 hover:to-blue-500 shadow-md shadow-cyan-500/20 transition-all"
                >
                  Save Deal
                </button>
              </div>
            </form>
          </div>
        </div>
      )}
    </div>
  );
};

export default Dashboard;
