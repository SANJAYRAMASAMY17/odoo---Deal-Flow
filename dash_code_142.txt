// Indian currency formatters (en-IN comma grouping & Lakh/Crore notation)
const formatINR = (val) => {
  if (val === undefined || val === null || isNaN(val)) return '₹0';
  return '₹' + Number(val).toLocaleString('en-IN');
};

const formatINRLakhCrore = (val) => {
  if (val === undefined || val === null || isNaN(val)) return '₹0';
  if (val >= 10000000) {
    return `₹${(val / 10000000).toFixed(2)} Cr`;
  }
  if (val >= 100000) {
    return `₹${(val / 100000).toFixed(2)} L`;
  }
  return '₹' + Number(val).toLocaleString('en-IN');
};

                    <td className="py-3.5 px-4 font-sans font-semibold text-white">Nova Retail India</td>
                    <td className="py-3.5 px-4 text-slate-400">2026-09-01</td>
                    <td className="py-3.5 px-4">₹9,75,000</td>
                    <td className="py-3.5 px-4 text-slate-300">₹1,75,500</td>
                    <td className="py-3.5 px-4 font-bold text-white">₹11,50,500</td>
                    <td className="py-3.5 px-4"><span className="px-2 py-0.5 rounded-full text-xs font-sans bg-amber-500/10 text-amber-400 border border-amber-500/20">Pending</span></td>
                  </tr>
                </tbody>
              </table>
            </div>
          </div>
        )}

        {/* =========================================================================
            MODULE 7: DEAL HEALTH
           ========================================================================= */}
        {activeModule === 'Deal Health' && (
          <div className="glass-card rounded-3xl p-6 sm:p-8 border border-slate-700/60 shadow-xl space-y-6">
            <div className="flex items-center justify-between border-b border-slate-800 pb-4">
              <div>
                <h2 className="text-2xl font-bold font-display text-white">Deal Health & AI Risk Scoring</h2>
                <p className="text-xs text-slate-400 mt-0.5">Automated drop-off flags, stalled velocity alerts, and recovery playbooks</p>
              </div>
              <span className="px-3 py-1 rounded-full text-xs font-semibold bg-rose-500/15 text-rose-300 border border-rose-500/30">
                3 Flagged by Deal Health
              </span>
            </div>

            <div className="space-y-4">
              <div className="p-5 rounded-2xl bg-rose-500/10 border border-rose-500/20 space-y-2">
                <div className="flex items-center justify-between">
                  <span className="text-sm font-bold text-white">Beta Industries (Pune) — Quotation QT-1003</span>
                  <span className="text-xs font-bold text-rose-400">Health Score: 48/100 (High Risk)</span>
                </div>
                <p className="text-xs text-rose-200/90">
                  <strong className="text-white">Flag Factor:</strong> Client requested 15% discount concession for Pune manufacturing plant; awaiting VP response over 48h.
                </p>
                <div className="pt-2 flex items-center justify-between text-xs text-slate-400">
                  <span>Recommended Action: Fast-track VP approval & offer volume tier</span>
                  <button onClick={() => advanceQuotationStage('QT-1003', 'Approved')} className="text-amber-400 hover:underline font-semibold">
                    Approve & Save Deal →
                  </button>
                </div>
              </div>

              <div className="p-5 rounded-2xl bg-amber-500/10 border border-amber-500/20 space-y-2">
                <div className="flex items-center justify-between">
                  <span className="text-sm font-bold text-white">Zenith Infotech (Hyderabad) — Quotation QT-1005</span>
                  <span className="text-xs font-bold text-amber-400">Health Score: 62/100 (Medium Risk)</span>
                </div>
                <p className="text-xs text-amber-200/90">
                  <strong className="text-white">Flag Factor:</strong> Legal redline revision stalled in procurement for 12 days without scheduled follow-up call.
                </p>
                <div className="pt-2 flex items-center justify-between text-xs text-slate-400">
                  <span>Recommended Action: Schedule touch-base with VP Procurement</span>
                  <button onClick={() => advanceQuotationStage('QT-1005', 'Confirmed')} className="text-amber-400 hover:underline font-semibold">
                    Confirm Terms →
                  </button>
                </div>
              </div>
            </div>
          </div>
        )}

        {/* =========================================================================
            MODULE 8: REPORTS
           ========================================================================= */}
        {activeModule === 'Reports' && (
          <div className="glass-card rounded-3xl p-6 sm:p-8 border border-slate-700/60 shadow-xl space-y-6">
            <div className="border-b border-slate-800 pb-4">
              <h2 className="text-2xl font-bold font-display text-white">Executive Sales & GST Analytics</h2>
              <p className="text-xs text-slate-400 mt-0.5">National pipeline conversion pacing, quarterly targets, and revenue forecasts</p>
            </div>

            <div className="grid grid-cols-1 md:grid-cols-3 gap-5">
              <div className="p-5 rounded-2xl bg-slate-900/70 border border-slate-800">
                <span className="text-xs text-slate-400 uppercase tracking-wider">FY26 Target Pacing</span>
                <div className="text-2xl font-extrabold text-white mt-1">₹4.80 Cr / ₹5.50 Cr (87.2%)</div>
                <div className="w-full bg-slate-800 rounded-full h-2 mt-3 overflow-hidden">
                  <div className="bg-gradient-to-r from-amber-400 to-emerald-500 h-full rounded-full" style={{ width: '87.2%' }} />
                </div>
              </div>
              <div className="p-5 rounded-2xl bg-slate-900/70 border border-slate-800">
                <span className="text-xs text-slate-400 uppercase tracking-wider">Average Deal Size (India)</span>
                <div className="text-2xl font-extrabold text-white mt-1">₹24,80,000</div>
                <p className="text-xs text-emerald-400 mt-2">+18.4% YoY B2B enterprise growth</p>
              </div>
              <div className="p-5 rounded-2xl bg-slate-900/70 border border-slate-800">
                <span className="text-xs text-slate-400 uppercase tracking-wider">Avg Sales Cycle</span>
                <div className="text-2xl font-extrabold text-white mt-1">14 Days</div>
                <p className="text-xs text-sky-400 mt-2">Accelerated with e-Signatures</p>
              </div>
            </div>
          </div>
        )}

        {/* =========================================================================
            MODULE 9: PRODUCT (with HSN & GST Tiers)
           ========================================================================= */}
        {activeModule === 'Product' && (
          <div className="glass-card rounded-3xl p-6 sm:p-8 border border-slate-700/60 shadow-xl space-y-6">
            <div className="border-b border-slate-800 pb-4">
              <h2 className="text-2xl font-bold font-display text-white">Product Catalog & HSN Code Master</h2>
              <p className="text-xs text-slate-400 mt-0.5">Sellable enterprise software tiers, hardware appliances, and GST tax codes</p>
            </div>

            <div className="grid grid-cols-1 md:grid-cols-3 gap-5">
              <div className="p-5 rounded-2xl bg-slate-900/70 border border-slate-800 space-y-2">
                <div className="flex items-center justify-between">
                  <span className="px-2.5 py-0.5 rounded-full text-[10px] font-semibold bg-sky-500/10 text-sky-400 border border-sky-500/20">SKU-CLD-100</span>
                  <span className="text-[10px] font-mono text-slate-400">HSN: 998313 (18% GST)</span>
                </div>
                <h4 className="text-base font-bold text-white">Cloud Core Platform Engine</h4>
                <p className="text-xs text-slate-400">Enterprise multi-region cloud orchestrator with automated failover across Indian data centers.</p>
                <div className="pt-2 text-sm font-mono font-bold text-amber-300">₹9,60,000 / seat tier</div>
              </div>

              <div className="p-5 rounded-2xl bg-slate-900/70 border border-slate-800 space-y-2">
                <div className="flex items-center justify-between">
                  <span className="px-2.5 py-0.5 rounded-full text-[10px] font-semibold bg-amber-500/10 text-amber-400 border border-amber-500/20">SKU-IOT-HUB</span>
                  <span className="text-[10px] font-mono text-slate-400">HSN: 851762 (18% GST)</span>
                </div>
                <h4 className="text-base font-bold text-white">Industrial IoT Telemetry Hub</h4>
                <p className="text-xs text-slate-400">Hardware appliance for edge computing & telemetry ingestion in manufacturing corridors.</p>
                <div className="pt-2 text-sm font-mono font-bold text-amber-300">₹4,20,000 / unit</div>
              </div>

              <div className="p-5 rounded-2xl bg-slate-900/70 border border-slate-800 space-y-2">
                <div className="flex items-center justify-between">
                  <span className="px-2.5 py-0.5 rounded-full text-[10px] font-semibold bg-emerald-500/10 text-emerald-400 border border-emerald-500/20">SKU-AUD-SOC2</span>
                  <span className="text-[10px] font-mono text-slate-400">HSN: 998313 (18% GST)</span>
                </div>
                <h4 className="text-base font-bold text-white">SOC 2 & RBI Compliance Engine</h4>
                <p className="text-xs text-slate-400">Real-time compliance monitoring and automated auditor evidence export.</p>
                <div className="pt-2 text-sm font-mono font-bold text-amber-300">₹12,50,000 / year</div>
              </div>
            </div>
          </div>
        )}
      </main>

      {/* =========================================================================
          MODAL: + NEW QUOTATION (GST COMPLIANT)
         ========================================================================= */}
      {isNewQuoteModalOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/75 backdrop-blur-md">
          <div className="relative w-full max-w-md glass-card rounded-3xl p-6 sm:p-8 border border-slate-700 shadow-2xl">
            <div className="flex items-center justify-between mb-6">
              <div className="flex items-center gap-2">
                <div className="w-8 h-8 rounded-lg bg-amber-500/20 text-amber-400 flex items-center justify-center">
                  <svg className="w-4 h-4" fill="none" stroke="currentColor" strokeWidth="2.5" viewBox="0 0 24 24">
                    <path strokeLinecap="round" strokeLinejoin="round" d="M12 4.5v15m7.5-7.5h-15" />
                  </svg>
                </div>
                <h3 className="text-xl font-bold font-display text-white">Create GST Quotation</h3>
              </div>
              <button
                onClick={() => setIsNewQuoteModalOpen(false)}
                className="p-1 rounded-lg text-slate-400 hover:text-white hover:bg-slate-800 transition-colors cursor-pointer"
              >
                <svg className="w-5 h-5" fill="none" stroke="currentColor" strokeWidth="2" viewBox="0 0 24 24">
                  <path strokeLinecap="round" strokeLinejoin="round" d="M6 18L18 6M6 6l12 12" />
                </svg>
              </button>
            </div>

            <form onSubmit={handleCreateQuotation} className="space-y-4">
              <div>
                <label className="block text-xs font-semibold text-slate-300 uppercase tracking-wider mb-1">
                  Company / Entity Name
                </label>
                <input
                  type="text"
                  value={newClient}
                  onChange={(e) => setNewClient(e.target.value)}
                  placeholder="e.g. Tata Advanced Systems Pvt Ltd"
                  required
                  className="w-full py-2.5 px-3.5 rounded-xl glass-input text-sm outline-none"
                />
              </div>

              <div className="grid grid-cols-2 gap-3">
                <div>
                  <label className="block text-xs font-semibold text-slate-300 uppercase tracking-wider mb-1">
                    State of Supply
                  </label>
                  <select
                    value={newCity}
                    onChange={(e) => setNewCity(e.target.value)}
                    className="w-full py-2.5 px-3 rounded-xl glass-input text-xs outline-none bg-slate-900 text-slate-200"
                  >
                    <option value="Bengaluru, Karnataka">Karnataka</option>
                    <option value="Mumbai, Maharashtra">Maharashtra</option>
                    <option value="Gurugram, Delhi-NCR">Delhi-NCR</option>
                    <option value="Hyderabad, Telangana">Telangana</option>
                    <option value="Chennai, Tamil Nadu">Tamil Nadu</option>
                    <option value="Ahmedabad, Gujarat">Gujarat</option>
                  </select>
                </div>

                <div>
                  <label className="block text-xs font-semibold text-slate-300 uppercase tracking-wider mb-1">
                    Client GSTIN
                  </label>
                  <input
                    type="text"
                    value={newGstin}
                    onChange={(e) => setNewGstin(e.target.value.toUpperCase())}
                    placeholder="29AABCA1234F1Z5"
                    className="w-full py-2.5 px-3.5 rounded-xl glass-input text-xs uppercase font-mono outline-none"
                  />
                </div>
              </div>

              <div>
                <label className="block text-xs font-semibold text-slate-300 uppercase tracking-wider mb-1">
                  Quotation Title / Scope
                </label>
                <input
                  type="text"
                  value={newTitle}
                  onChange={(e) => setNewTitle(e.target.value)}
                  placeholder="e.g. Enterprise Cloud & ABDM Gateway Integration"
                  className="w-full py-2.5 px-3.5 rounded-xl glass-input text-sm outline-none"
                />
              </div>

              <div className="grid grid-cols-2 gap-3">
                <div>
                  <label className="block text-xs font-semibold text-slate-300 uppercase tracking-wider mb-1">
                    Taxable Amount (₹ INR)
                  </label>
                  <input
                    type="number"
                    value={newAmount}
                    onChange={(e) => setNewAmount(e.target.value)}
                    placeholder="1250000"
                    required
                    className="w-full py-2.5 px-3.5 rounded-xl glass-input text-sm outline-none font-mono"
                  />
                </div>

                <div>
                  <label className="block text-xs font-semibold text-slate-300 uppercase tracking-wider mb-1">
                    GST Structure
                  </label>
                  <select
                    value={gstType}
                    onChange={(e) => setGstType(e.target.value)}
                    className="w-full py-2.5 px-3 rounded-xl glass-input text-xs outline-none bg-slate-900 text-slate-200"
                  >
                    <option value="IGST">18% IGST (Interstate)</option>
                    <option value="CGST_SGST">9% CGST + 9% SGST</option>
                  </select>
                </div>
              </div>

              {newAmount && (
                <div className="p-3 rounded-xl bg-slate-900/80 border border-slate-800 text-xs space-y-1 font-mono">
                  <div className="flex justify-between text-slate-400">
                    <span>Taxable Value:</span>
                    <span>{formatINR(parseFloat(newAmount))}</span>
                  </div>
                  <div className="flex justify-between text-amber-300">
                    <span>GST (18%):</span>
                    <span>{formatINR(parseFloat(newAmount) * 0.18)}</span>
                  </div>
                  <div className="flex justify-between text-white font-bold border-t border-slate-800 pt-1">
                    <span>Total (Incl. GST):</span>
                    <span className="text-emerald-400">{formatINR(parseFloat(newAmount) * 1.18)}</span>
                  </div>
                </div>
              )}

              <div className="pt-3 flex items-center justify-end gap-3">
                <button
                  type="button"
                  onClick={() => setIsNewQuoteModalOpen(false)}
                  className="px-4 py-2.5 rounded-xl text-xs font-medium text-slate-400 hover:text-slate-200 transition-colors cursor-pointer"
                >
                  Cancel
                </button>
                <button
                  type="submit"
                  className="px-5 py-2.5 rounded-xl text-xs font-semibold text-white bg-gradient-to-r from-amber-500 to-blue-600 hover:from-amber-400 hover:to-blue-500 shadow-lg shadow-amber-500/25 transition-all cursor-pointer"
                >
                  Generate GST Quotation
                </button>
              </div>
            </form>
          </div>
        </div>
      )}

      {/* =========================================================================
          MODAL: QUOTATION DETAIL INSPECTOR (with GST Breakdown & Indian Currency)
         ========================================================================= */}
      {selectedQuotation && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/75 backdrop-blur-md">
          <div className="relative w-full max-w-2xl glass-card rounded-3xl p-6 sm:p-8 border border-slate-700 shadow-2xl space-y-6 animate-float">
            {/* Header */}
            <div className="flex items-start justify-between border-b border-slate-800 pb-4">
              <div>
                <div className="flex items-center gap-2">
                  <span className="font-mono text-xs font-bold text-amber-400">{selectedQuotation.id}</span>
                  <span className={`px-2.5 py-0.5 rounded-full text-xs font-medium border ${getStageColor(selectedQuotation.stage)}`}>
                    {selectedQuotation.stage}
                  </span>
                  <span className="text-[10px] font-mono px-2 py-0.5 rounded-full bg-slate-800 text-slate-300">
                    GSTIN: {selectedQuotation.gstin || '29AABCT1332L1ZV'}
                  </span>
                </div>
                <h3 className="text-2xl font-extrabold font-display text-white mt-1">
                  {selectedQuotation.client}
                </h3>
                <p className="text-xs text-slate-400 mt-0.5">
                  {selectedQuotation.title} • {selectedQuotation.city || 'India'}
                </p>
              </div>

              <button
                onClick={() => setSelectedQuotation(null)}
                className="p-1.5 rounded-xl text-slate-400 hover:text-white hover:bg-slate-800 transition-colors cursor-pointer"
              >
                <svg className="w-5 h-5" fill="none" stroke="currentColor" strokeWidth="2" viewBox="0 0 24 24">
                  <path strokeLinecap="round" strokeLinejoin="round" d="M6 18L18 6M6 6l12 12" />
                </svg>
              </button>
            </div>

            {/* 5-Stage Stepper */}
            <div>
              <span className="text-[11px] font-semibold text-slate-400 uppercase tracking-wider block mb-2">Deal Stage Progression</span>
              <div className="grid grid-cols-5 gap-1.5">
                {kanbanStages.map((stage, idx) => {
                  const currentIdx = kanbanStages.indexOf(selectedQuotation.stage);
                  const isPassed = idx <= currentIdx;
                  const isCurrent = idx === currentIdx;

                  return (
                    <button
                      key={stage}
                      onClick={() => advanceQuotationStage(selectedQuotation.id, stage)}
                      className={`py-2 px-1 rounded-lg text-[10px] font-bold text-center transition-all cursor-pointer ${
                        isCurrent
                          ? 'bg-amber-500 text-white shadow-md shadow-amber-500/25 ring-1 ring-white/20'
                          : isPassed
                          ? 'bg-blue-600/30 text-blue-300 border border-blue-500/40'
                          : 'bg-slate-800/60 text-slate-500 border border-slate-800 hover:text-slate-300'
                      }`}
                    >
                      {stage}
                    </button>
                  );
                })}
              </div>
            </div>

            {/* Line Items & GST Computation */}
            <div className="p-4 rounded-2xl bg-slate-900/60 border border-slate-800 space-y-3">
              <span className="text-xs font-semibold text-slate-300 uppercase tracking-wider block">Scope of Work & HSN Itemization</span>
              <div className="divide-y divide-slate-800 text-xs">
                {selectedQuotation.items ? (
                  selectedQuotation.items.map((item, i) => (
                    <div key={i} className="py-2 flex items-center justify-between">
                      <div>
                        <span className="text-slate-200 font-medium">{item.desc} (Qty: {item.qty})</span>
                        <span className="text-[10px] text-slate-400 block font-mono">HSN: {item.hsn || '998313'}</span>
                      </div>
                      <span className="font-mono font-semibold text-white">{formatINR(item.rate * item.qty)}</span>
                    </div>
                  ))
                ) : (
                  <div className="py-2 flex items-center justify-between">
                    <div>
                      <span className="text-slate-200 font-medium">{selectedQuotation.title}</span>
                      <span className="text-[10px] text-slate-400 block font-mono">HSN: {selectedQuotation.hsn || '998313'}</span>
                    </div>
                    <span className="font-mono font-semibold text-white">{formatINR(selectedQuotation.amount)}</span>
                  </div>
                )}
              </div>

              {/* GST Calculations */}
              <div className="pt-2 border-t border-slate-800 space-y-1 text-xs font-mono">
                <div className="flex justify-between text-slate-400">
                  <span>Taxable Subtotal:</span>
                  <span>{formatINR(selectedQuotation.amount)}</span>
                </div>
                <div className="flex justify-between text-amber-300">
                  <span>Integrated GST (IGST @ 18%):</span>
                  <span>{formatINR(selectedQuotation.amount * 0.18)}</span>
                </div>
                <div className="flex justify-between text-sm font-bold border-t border-slate-700/80 pt-1.5">
                  <span className="text-white">Total Commercial Amount:</span>
                  <span className="text-emerald-400 text-base font-extrabold">
                    {formatINR(selectedQuotation.amount * 1.18)}
                  </span>
                </div>
              </div>
            </div>

            {/* Notes */}
            {selectedQuotation.notes && (
              <p className="text-xs text-slate-400 italic bg-slate-900/40 p-3 rounded-xl border border-slate-800/80">
                "{selectedQuotation.notes}"
              </p>
            )}

            {/* Actions */}
            <div className="pt-2 flex items-center justify-end gap-3">
              <button
                onClick={() => setSelectedQuotation(null)}
                className="px-4 py-2 rounded-xl text-xs font-medium text-slate-400 hover:text-slate-200 transition-colors"
              >
                Close Inspector
              </button>

              {selectedQuotation.stage === 'Draft' && (
                <button
                  onClick={() => advanceQuotationStage(selectedQuotation.id, 'Pending Approval')}
                  className="px-4 py-2 rounded-xl text-xs font-semibold text-white bg-amber-600 hover:bg-amber-500 shadow-md transition-all cursor-pointer"
                >
                  Submit for Approval →
                </button>
              )}

              {selectedQuotation.stage === 'Pending Approval' && (
                <button
                  onClick={() => advanceQuotationStage(selectedQuotation.id, 'Approved')}
                  className="px-4 py-2 rounded-xl text-xs font-semibold text-white bg-emerald-600 hover:bg-emerald-500 shadow-md transition-all cursor-pointer"
                >
                  ✓ Executive Approval
                </button>
              )}

              {selectedQuotation.stage === 'Approved' && (
                <button
                  onClick={() => advanceQuotationStage(selectedQuotation.id, 'Negotiation')}
                  className="px-4 py-2 rounded-xl text-xs font-semibold text-white bg-indigo-600 hover:bg-indigo-500 shadow-md transition-all cursor-pointer"
                >
                  Send for Redline Negotiation →
                </button>
              )}

              {selectedQuotation.stage === 'Negotiation' && (
                <button
                  onClick={() => advanceQuotationStage(selectedQuotation.id, 'Confirmed')}
                  className="px-4 py-2 rounded-xl text-xs font-semibold text-white bg-sky-600 hover:bg-sky-500 shadow-md transition-all cursor-pointer"
                >
                  Confirm & Generate e-Invoice →
                </button>
              )}
            </div>
          </div>
        </div>
      )}
    </div>
  );
};

export default Dashboard;
