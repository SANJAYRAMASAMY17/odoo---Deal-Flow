          <div className="glass-card rounded-3xl p-6 sm:p-8 border border-slate-700/60 shadow-xl space-y-6">
            <div className="border-b border-slate-800 pb-4">
              <h2 className="text-2xl font-bold font-display text-white">Executive Sales Reports</h2>
              <p className="text-xs text-slate-400 mt-0.5">Pipeline conversion pacing, quarterly targets, and revenue forecasts</p>
            </div>

            <div className="grid grid-cols-1 md:grid-cols-3 gap-5">
              <div className="p-5 rounded-2xl bg-slate-900/70 border border-slate-800">
                <span className="text-xs text-slate-400 uppercase tracking-wider">Q3 Target Pacing</span>
                <div className="text-2xl font-extrabold text-white mt-1">88.4%</div>
                <div className="w-full bg-slate-800 rounded-full h-2 mt-3 overflow-hidden">
                  <div className="bg-gradient-to-r from-cyan-400 to-blue-500 h-full rounded-full" style={{ width: '88.4%' }} />
                </div>
              </div>
              <div className="p-5 rounded-2xl bg-slate-900/70 border border-slate-800">
                <span className="text-xs text-slate-400 uppercase tracking-wider">Average Deal Size</span>
                <div className="text-2xl font-extrabold text-white mt-1">$24,850</div>
                <p className="text-xs text-emerald-400 mt-2">+14.2% YoY expansion</p>
              </div>
              <div className="p-5 rounded-2xl bg-slate-900/70 border border-slate-800">
                <span className="text-xs text-slate-400 uppercase tracking-wider">Quotation-to-Close Cycle</span>
                <div className="text-2xl font-extrabold text-white mt-1">11.2 Days</div>
                <p className="text-xs text-cyan-400 mt-2">Accelerated by 3.5 days</p>
              </div>
            </div>
          </div>
        )}

        {/* =========================================================================
            MODULE 9: PRODUCT
           ========================================================================= */}
        {activeModule === 'Product' && (
          <div className="glass-card rounded-3xl p-6 sm:p-8 border border-slate-700/60 shadow-xl space-y-6">
            <div className="border-b border-slate-800 pb-4">
              <h2 className="text-2xl font-bold font-display text-white">Product Catalog & Price Books</h2>
              <p className="text-xs text-slate-400 mt-0.5">Sellable enterprise software tiers, hardware appliances, and add-on modules</p>
            </div>

            <div className="grid grid-cols-1 md:grid-cols-3 gap-5">
              <div className="p-5 rounded-2xl bg-slate-900/70 border border-slate-800 space-y-2">
                <span className="px-2.5 py-0.5 rounded-full text-[10px] font-semibold bg-cyan-500/10 text-cyan-400 border border-cyan-500/20">SKU-CLD-100</span>
                <h4 className="text-base font-bold text-white">Cloud Core Platform</h4>
                <p className="text-xs text-slate-400">Enterprise multi-region cloud orchestrator with automated failover.</p>
                <div className="pt-2 text-sm font-mono font-bold text-cyan-300">$9,600 / seat tier</div>
              </div>

              <div className="p-5 rounded-2xl bg-slate-900/70 border border-slate-800 space-y-2">
                <span className="px-2.5 py-0.5 rounded-full text-[10px] font-semibold bg-indigo-500/10 text-indigo-400 border border-indigo-500/20">SKU-IOT-HUB</span>
                <h4 className="text-base font-bold text-white">Industrial IoT Hub Appliance</h4>
                <p className="text-xs text-slate-400">Hardware gateway unit for telemetry streaming and sensor edge computing.</p>
                <div className="pt-2 text-sm font-mono font-bold text-cyan-300">$4,200 / unit</div>
              </div>

              <div className="p-5 rounded-2xl bg-slate-900/70 border border-slate-800 space-y-2">
                <span className="px-2.5 py-0.5 rounded-full text-[10px] font-semibold bg-emerald-500/10 text-emerald-400 border border-emerald-500/20">SKU-AUD-SOC2</span>
                <h4 className="text-base font-bold text-white">SOC 2 Automated Audit Engine</h4>
                <p className="text-xs text-slate-400">Real-time compliance monitoring and automated auditor evidence exports.</p>
                <div className="pt-2 text-sm font-mono font-bold text-cyan-300">$12,500 / year</div>
              </div>
            </div>
          </div>
        )}
      </main>

      {/* =========================================================================
          MODAL: + NEW QUOTATION CREATOR
         ========================================================================= */}
      {isNewQuoteModalOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/75 backdrop-blur-md">
          <div className="relative w-full max-w-md glass-card rounded-3xl p-6 sm:p-8 border border-slate-700 shadow-2xl">
            <div className="flex items-center justify-between mb-6">
              <div className="flex items-center gap-2">
                <div className="w-8 h-8 rounded-lg bg-cyan-500/20 text-cyan-400 flex items-center justify-center">
                  <svg className="w-4 h-4" fill="none" stroke="currentColor" strokeWidth="2.5" viewBox="0 0 24 24">
                    <path strokeLinecap="round" strokeLinejoin="round" d="M12 4.5v15m7.5-7.5h-15" />
                  </svg>
                </div>
                <h3 className="text-xl font-bold font-display text-white">Create New Quotation</h3>
              </div>
              <button
                onClick={() => setIsNewQuoteModalOpen(false)}
                className="p-1 rounded-lg text-slate-400 hover:text-white hover:bg-slate-800 transition-colors cursor-pointer"
              >
                <svg className="w-5 h-5" fill="none" stroke="currentColor" strokeWidth="2" viewBox="0 0 24 24">
                  <path strokeLinecap="round" strokeLinejoin="round" d="M6 18L18 6M6 6l12 12" />
                </svg>
              </button>
            </div>

            <form onSubmit={handleCreateQuotation} className="space-y-4">
              <div>
                <label className="block text-xs font-semibold text-slate-300 uppercase tracking-wider mb-1">
                  Client / Organization
                </label>
                <input
                  type="text"
                  value={newClient}
                  onChange={(e) => setNewClient(e.target.value)}
                  placeholder="e.g. Zenith Aerospace"
                  required
                  className="w-full py-2.5 px-3.5 rounded-xl glass-input text-sm outline-none"
                />
              </div>

              <div>
                <label className="block text-xs font-semibold text-slate-300 uppercase tracking-wider mb-1">
                  Quotation Title / Scope
                </label>
                <input
                  type="text"
                  value={newTitle}
                  onChange={(e) => setNewTitle(e.target.value)}
                  placeholder="e.g. Cloud Security Integration"
                  className="w-full py-2.5 px-3.5 rounded-xl glass-input text-sm outline-none"
                />
              </div>

              <div className="grid grid-cols-2 gap-3">
                <div>
                  <label className="block text-xs font-semibold text-slate-300 uppercase tracking-wider mb-1">
                    Value ($ USD)
                  </label>
                  <input
                    type="number"
                    value={newAmount}
                    onChange={(e) => setNewAmount(e.target.value)}
                    placeholder="18500"
                    required
                    className="w-full py-2.5 px-3.5 rounded-xl glass-input text-sm outline-none"
                  />
                </div>

                <div>
                  <label className="block text-xs font-semibold text-slate-300 uppercase tracking-wider mb-1">
                    Initial Stage
                  </label>
                  <select
                    value={newStage}
                    onChange={(e) => setNewStage(e.target.value)}
                    className="w-full py-2.5 px-3.5 rounded-xl glass-input text-sm outline-none bg-slate-900 text-slate-200"
                  >
                    <option value="Draft">Draft</option>
                    <option value="Pending Approval">Pending Approval</option>
                    <option value="Negotiation">Negotiation</option>
                  </select>
                </div>
              </div>

              <div className="pt-3 flex items-center justify-end gap-3">
                <button
                  type="button"
                  onClick={() => setIsNewQuoteModalOpen(false)}
                  className="px-4 py-2.5 rounded-xl text-xs font-medium text-slate-400 hover:text-slate-200 transition-colors cursor-pointer"
                >
                  Cancel
                </button>
                <button
                  type="submit"
                  className="px-5 py-2.5 rounded-xl text-xs font-semibold text-white bg-gradient-to-r from-cyan-500 to-blue-600 hover:from-cyan-400 hover:to-blue-500 shadow-lg shadow-cyan-500/25 transition-all cursor-pointer"
                >
                  Add Quotation
                </button>
              </div>
            </form>
          </div>
        </div>
      )}

      {/* =========================================================================
          MODAL / DRAWER: QUOTATION DETAIL INSPECTOR ("click a row to open it")
         ========================================================================= */}
      {selectedQuotation && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/75 backdrop-blur-md">
          <div className="relative w-full max-w-2xl glass-card rounded-3xl p-6 sm:p-8 border border-slate-700 shadow-2xl space-y-6 animate-float">
            {/* Header */}
            <div className="flex items-start justify-between border-b border-slate-800 pb-4">
              <div>
                <div className="flex items-center gap-2">
                  <span className="font-mono text-xs font-bold text-cyan-400">{selectedQuotation.id}</span>
                  <span className={`px-2.5 py-0.5 rounded-full text-xs font-medium border ${getStageColor(selectedQuotation.stage)}`}>
                    {selectedQuotation.stage}
                  </span>
                </div>
                <h3 className="text-2xl font-extrabold font-display text-white mt-1">
                  {selectedQuotation.client}
                </h3>
                <p className="text-xs text-slate-400 mt-0.5">
                  {selectedQuotation.title}
                </p>
              </div>

              <button
                onClick={() => setSelectedQuotation(null)}
                className="p-1.5 rounded-xl text-slate-400 hover:text-white hover:bg-slate-800 transition-colors cursor-pointer"
              >
                <svg className="w-5 h-5" fill="none" stroke="currentColor" strokeWidth="2" viewBox="0 0 24 24">
                  <path strokeLinecap="round" strokeLinejoin="round" d="M6 18L18 6M6 6l12 12" />
                </svg>
              </button>
            </div>

            {/* 5-Stage Visual Stepper */}
            <div>
              <span className="text-[11px] font-semibold text-slate-400 uppercase tracking-wider block mb-2">Stage Progression</span>
              <div className="grid grid-cols-5 gap-1.5">
                {kanbanStages.map((stage, idx) => {
                  const currentIdx = kanbanStages.indexOf(selectedQuotation.stage);
                  const isPassed = idx <= currentIdx;
                  const isCurrent = idx === currentIdx;

                  return (
                    <button
                      key={stage}
                      onClick={() => advanceQuotationStage(selectedQuotation.id, stage)}
                      className={`py-2 px-1 rounded-lg text-[10px] font-bold text-center transition-all cursor-pointer ${
                        isCurrent
                          ? 'bg-cyan-500 text-white shadow-md shadow-cyan-500/25 ring-1 ring-white/20'
                          : isPassed
                          ? 'bg-blue-600/30 text-blue-300 border border-blue-500/40'
                          : 'bg-slate-800/60 text-slate-500 border border-slate-800 hover:text-slate-300'
                      }`}
                    >
                      {stage}
                    </button>
                  );
                })}
              </div>
            </div>

            {/* Line Items Breakdown */}
            <div className="p-4 rounded-2xl bg-slate-900/60 border border-slate-800 space-y-3">
              <span className="text-xs font-semibold text-slate-300 uppercase tracking-wider block">Scope of Work / Line Items</span>
              <div className="divide-y divide-slate-800 text-xs">
                {selectedQuotation.items ? (
                  selectedQuotation.items.map((item, i) => (
                    <div key={i} className="py-2 flex items-center justify-between">
                      <span className="text-slate-300">{item.desc} (Qty: {item.qty})</span>
                      <span className="font-mono font-semibold text-white">${(item.rate * item.qty).toLocaleString()}</span>
                    </div>
                  ))
                ) : (
                  <div className="py-2 flex items-center justify-between">
                    <span className="text-slate-300">{selectedQuotation.title}</span>
                    <span className="font-mono font-semibold text-white">${selectedQuotation.amount.toLocaleString()}</span>
                  </div>
                )}
              </div>

              <div className="pt-2 border-t border-slate-800 flex items-center justify-between text-sm">
                <span className="font-bold text-white">Total Commercial Value</span>
                <span className="font-mono font-extrabold text-cyan-400 text-lg">
                  ${selectedQuotation.amount.toLocaleString()} USD
                </span>
              </div>
            </div>

            {/* Notes */}
            {selectedQuotation.notes && (
              <p className="text-xs text-slate-400 italic bg-slate-900/40 p-3 rounded-xl border border-slate-800/80">
                "{selectedQuotation.notes}"
              </p>
            )}

            {/* Actions */}
            <div className="pt-2 flex items-center justify-end gap-3">
              <button
                onClick={() => setSelectedQuotation(null)}
                className="px-4 py-2 rounded-xl text-xs font-medium text-slate-400 hover:text-slate-200 transition-colors"
              >
                Close Inspector
              </button>

              {selectedQuotation.stage === 'Draft' && (
                <button
                  onClick={() => advanceQuotationStage(selectedQuotation.id, 'Pending Approval')}
                  className="px-4 py-2 rounded-xl text-xs font-semibold text-white bg-amber-600 hover:bg-amber-500 shadow-md transition-all cursor-pointer"
                >
                  Submit for Approval →
                </button>
              )}

              {selectedQuotation.stage === 'Pending Approval' && (
                <button
                  onClick={() => advanceQuotationStage(selectedQuotation.id, 'Approved')}
                  className="px-4 py-2 rounded-xl text-xs font-semibold text-white bg-emerald-600 hover:bg-emerald-500 shadow-md transition-all cursor-pointer"
                >
                  ✓ Executive Approval
                </button>
              )}

              {selectedQuotation.stage === 'Approved' && (
                <button
                  onClick={() => advanceQuotationStage(selectedQuotation.id, 'Negotiation')}
                  className="px-4 py-2 rounded-xl text-xs font-semibold text-white bg-indigo-600 hover:bg-indigo-500 shadow-md transition-all cursor-pointer"
                >
                  Send to Client for Negotiation →
                </button>
              )}

              {selectedQuotation.stage === 'Negotiation' && (
                <button
                  onClick={() => advanceQuotationStage(selectedQuotation.id, 'Confirmed')}
                  className="px-4 py-2 rounded-xl text-xs font-semibold text-white bg-cyan-600 hover:bg-cyan-500 shadow-md transition-all cursor-pointer"
                >
                  Confirm & Generate Invoice →
                </button>
              )}
            </div>
          </div>
        </div>
      )}
    </div>
  );
};

export default Dashboard;
