    const utrNumber = `HDFC-UTR-${getNextActivityId()}910`;
    const updated = {
      ...matched,
      status: 'Paid',
      utr: utrNumber,
      paymentMode: 'NEFT / RTGS (Direct Settlement)',
    };

    setInvoices((prev) => prev.map((inv) => (inv.id === invoiceId ? updated : inv)));
    if (selectedInvoiceDetail && selectedInvoiceDetail.id === invoiceId) {
      setSelectedInvoiceDetail(updated);
    }

    setActivities((prev) => [
      {
        id: getNextActivityId(),
        title: `Payment recorded for ${matched.customer} (${invoiceId}) • ${utrNumber}`,
        category: 'Invoice Settlement',
        time: 'Just now',
        type: 'success',
        badge: 'Paid',
      },
      ...prev,
    ]);

    showNotification(`Invoice ${invoiceId} marked as Paid! UTR: ${utrNumber}`);
  };

    setActivities((prev) => [
      {
        id: getNextActivityId(),
        title: `Payment reminder dispatched for ${matched.customer} (${invoiceId})`,
        category: 'Accounts Receivable',
        time: 'Just now',
        type: 'warning',
        badge: 'Reminder Sent',
      },
      ...prev,
    ]);

    showNotification(`Automated payment reminder sent to ${matched.customer} finance team!`);
  };
